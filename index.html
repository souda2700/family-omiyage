// スプレッドシートからデータを取得または追加・更新するGAS処理（日本語ヘッダー対応版） function doGet(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = sheet.getDataRange().getValues(); data.shift(); // 1行目（日本語ヘッダー）を取り除く const familyId = e.parameter.family_id; // 家族の合言葉（B列：家族ID）に一致するデータのみ抽出 const result = data .filter(row => row[1] && row[1].toString() === familyId) .map(row => ({ id: row[0], // A列: ID family_id: row[1], // B列: 家族ID recipient: row[2], // C列: 渡す相手 category: row[3], // D列: ジャンル item_name: row[4], // E列: 品物・メモ is_bought: row[5] === true || row[5] === "TRUE" // F列: 購入済みフラグ })); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const params = JSON.parse(e.postData.contents); const action = params.action; if (action === "add") { // 新規お土産の追加 const newId = new Date().getTime().toString(); // 簡易的なユニークID sheet.appendRow([ newId, params.family_id, params.recipient, params.category, params.item_name, false ]); return ContentService.createTextOutput(JSON.stringify({ status: "success", id: newId })) .setMimeType(ContentService.MimeType.JSON); } else if (action === "toggle") { // 購入状態（購入済み/未購入）の切り替え const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0].toString() === params.id.toString()) { const currentStatus = data[i][5]; sheet.getRange(i + 1, 6).setValue(!currentStatus); break; } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } }
